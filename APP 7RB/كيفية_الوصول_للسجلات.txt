📍 الطريقة الأولى: من خلال صفحة الويب (الأسهل)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. افتح PowerShell أو Command Prompt
2. اذهب إلى مجلد المشروع:
   cd "C:\Users\pc\Desktop\APP 7RB"

3. ثبت التبعيات (مرة واحدة فقط):
   npm install

4. شغل الخادم:
   npm start

5. افتح المتصفح واذهب إلى:
   http://localhost:3000/logs-viewer

   هنا ستشوف جميع السجلات:
   ✅ من دخل الموقع
   ✅ من سجل حساب جديد
   ✅ من سجل دخول
   ✅ جميع الأنشطة

═══════════════════════════════════════════════════════════

📍 الطريقة الثانية: من الملفات مباشرة
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

بعد تشغيل الخادم، السجلات تُحفظ في:
C:\Users\pc\Desktop\APP 7RB\logs\

كل يوم له ملف منفصل:
- bloodhunt-2026-01-22.log (اليوم)
- bloodhunt-2026-01-21.log (أمس)
- وهكذا...

يمكنك فتح هذه الملفات بأي محرر نصوص لمشاهدة السجلات.

═══════════════════════════════════════════════════════════

📍 الطريقة الثالثة: من Console المتصفح
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

إذا لم يكن الخادم يعمل، السجلات تُحفظ في localStorage:

1. افتح المتصفح
2. اضغط F12 لفتح Developer Tools
3. اذهب إلى Console
4. اكتب:
   JSON.parse(localStorage.getItem('bloodHuntLogs'))

ستشوف جميع السجلات المحفوظة محلياً.

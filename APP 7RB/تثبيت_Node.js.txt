   كيفية تثبيت Node.js لتشغيل نظام السجلات
═══════════════════════════════════════════════════════════

📥 خطوات التثبيت:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. اذهب إلى الموقع الرسمي:
   https://nodejs.org/

2. حمّل النسخة LTS (الموصى بها)
   - اضغط على "Download for Windows"

3. شغل الملف المحمّل واتبع التعليمات
   - اضغط Next في كل الخطوات
   - تأكد من اختيار "Add to PATH"

4. بعد التثبيت، أعد تشغيل PowerShell أو Command Prompt

5. تحقق من التثبيت:
   node --version
   npm --version

6. بعدها شغل الأوامر:
   cd "C:\Users\pc\Desktop\APP 7RB"
   npm install
   npm start

═══════════════════════════════════════════════════════════

🚀 بعد التثبيت:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. افتح PowerShell في مجلد المشروع
2. npm install (للمرة الأولى فقط)
3. npm start (لتشغيل الخادم)
4. افتح المتصفح: http://localhost:3000/logs-viewer
